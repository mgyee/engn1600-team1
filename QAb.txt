** kN SWEEP

.param kWECLKb=1
.param kWECLK=1
.param kWEMb=1
.param kQb=1
.param kN=1

.control

** Define input signals
let f = 1e8
let T = 1/f
let PW = T/2
let INF = 5 * T

let tstop = 4 * T
let tstep = 0.001 * T
let NTRIALS = 30

compose kVALS start=1 stop=5 lin=$&NTRIALS
compose TRISE start=0 stop=0 lin=$&NTRIALS
compose TFALL start=0 stop=0 lin=$&NTRIALS

let idx = 0
while idx < NTRIALS
	let kVAL = kVALS[idx]
	alterparam kQb = $&kVAL
	reset

	** Enable READ_A
	alter @VRA[PULSE] = [ 0 3.3 4n 1p 1p $&PW 20n 0 ]

	** Disable READ_B
	alter @VRB[DC] = [ 0 3.3 9n 1p 1p $&PW 50n 0 ]
	* [ 0 3.3 0 1p 1p $&PW 30n 0 ]

	** Test N
	alter @VN[PULSE] = [ 3.3 0 4n 1p 1p $&PW 50n 0 ]

	** Test WECLK
	alter @VCLK1[PULSE] = [ 0 3.3 $&PW 1p 1p $&PW $&T 0 ]
	
	** Test WECLKb
	alter @VCLK2[PULSE] = [ 3.3 0 $&PW 1p 1p $&PW $&T 0 ]


	** Test pulses
	alter @VD[PWL] = [ 0 0 $&PW 0 $&PW 3.3 $&T 3.3 $&T 0 ]

	tran $&tstep $&tstop
	meas tran TPLH TRIG V(RA) VAL=1.65 RISE=2 TARG V(QAb) VAL=1.65 RISE=1 TD=22n
	meas tran TPHL TRIG V(RA) VAL=1.65 FALL=1 TARG V(QAb) VAL=1.65 FALL=1
	let TRISE[idx] = $&TPLH
	let TFALL[idx] = $&TPHL
	let idx = idx + 1
end

plot TRISE vs kVALS
plot TFALL vs kVALS

let minval = minimum(TRISE)
let idxR = 0
while TRISE[idxR] > minval
  let idxR = idxR + 1
end
print kVALS[idxR]

let minval = minimum(TFALL)
let idxF = 0
while TFALL[idxF] > minval
  let idxF = idxF + 1
end
print kVALS[idxF]

.endc
